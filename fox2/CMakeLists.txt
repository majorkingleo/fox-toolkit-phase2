cmake_minimum_required(VERSION 3.16)

# Version of the code (keep in sync with configure.ac)
project(FOX VERSION 2.0.0 LANGUAGES CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "In-source builds are not supported; use: cmake -S . -B <builddir>")
endif()

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")

# 64-bit Unix builds install libraries in lib64 unless the caller chose a libdir
if(UNIX AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT DEFINED CMAKE_INSTALL_LIBDIR)
  set(CMAKE_INSTALL_LIBDIR lib64 CACHE PATH "Object code libraries")
endif()
include(GNUInstallDirs)
include(FOXFeatures)

# The Wayland protocol glue generated by wayland-scanner is C
if(FOX_PLATFORM STREQUAL "wayland")
  enable_language(C)
endif()

set(FOX_MAJOR_VERSION ${PROJECT_VERSION_MAJOR})
set(FOX_MINOR_VERSION ${PROJECT_VERSION_MINOR})
set(FOX_PATCH_LEVEL   ${PROJECT_VERSION_PATCH})
configure_file(include/fxver.h.in include/fxver.h @ONLY)

# Linker flags for the fox-config script and the pkg-config file
function(fox_flag_string OUTVAR)
  set(_flags "")
  foreach(_lib ${ARGN})
    if(_lib STREQUAL "Threads::Threads")
      string(APPEND _flags " ${CMAKE_THREAD_LIBS_INIT}")
    elseif(_lib MATCHES "^-")
      string(APPEND _flags " ${_lib}")
    elseif(IS_ABSOLUTE "${_lib}")
      get_filename_component(_name "${_lib}" NAME_WE)
      string(REGEX REPLACE "^lib" "" _name "${_name}")
      string(APPEND _flags " -l${_name}")
    else()
      string(APPEND _flags " -l${_lib}")
    endif()
  endforeach()
  string(STRIP "${_flags}" _flags)
  set(${OUTVAR} "${_flags}" PARENT_SCOPE)
endfunction()
set(FOX_PC_ALL_LIBS ${FOX_PUBLIC_LIBRARIES} ${FOX_PRIVATE_LIBRARIES})
list(REMOVE_DUPLICATES FOX_PC_ALL_LIBS)
fox_flag_string(FOX_PC_LINK_LIBS ${FOX_PC_ALL_LIBS})
set(PLATFORM_LIBS "${FOX_PC_LINK_LIBS}")
set(GL_LIBS "")
set(LIBS "")
set(prefix "${CMAKE_INSTALL_PREFIX}")
set(exec_prefix "\${prefix}")
set(libdir "\${exec_prefix}/${CMAKE_INSTALL_LIBDIR}")
set(includedir "\${prefix}/${CMAKE_INSTALL_INCLUDEDIR}")
configure_file(fox-2.0.pc.in "${CMAKE_BINARY_DIR}/fox-2.0.pc" @ONLY)
configure_file(fox-config.in "${CMAKE_BINARY_DIR}/fox-config" @ONLY)

# Executables and the DLL go to one directory, so tests find the DLL on Windows
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")

add_subdirectory(utils)
add_subdirectory(lib)

if(FOX_BUILD_TESTS)
  add_subdirectory(tools)
  enable_testing()
  add_subdirectory(apps)
  add_subdirectory(tests)
endif()

install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/fox-2.0
        FILES_MATCHING PATTERN "*.h")
install(FILES "${CMAKE_BINARY_DIR}/include/fxver.h"
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/fox-2.0)
install(PROGRAMS "${CMAKE_BINARY_DIR}/fox-config"
        DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES "${CMAKE_BINARY_DIR}/fox-2.0.pc"
        DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
